# Renfield Satellite Requirements
# For Raspberry Pi Zero 2 W and Orange Pi Zero 3W (64-bit, aarch64)
#
# The OTA installs this file into the satellite's venv (`pip install --no-deps`),
# so it must name exactly what the fleet runs: the packages the playbook installs
# (group_vars `satellite_python_packages` + its dedicated pip tasks). A package
# listed here but not on the fleet would be installed by the next OTA — that is
# how pyopen-wakeword would have switched the wake word backend.
# tests/satellite/test_requirements_match_fleet.py holds the two lists together.
#
# Inspired by OHF-Voice/linux-voice-assistant

# WebSocket communication
websockets>=12.0

# HTTP client for model downloads
aiohttp>=3.9.0

# Audio capture
pyaudio>=0.2.13
numpy>=1.24.0

# Audio preprocessing (optional but recommended for better recognition)
noisereduce>=3.0.0  # Spectral noise reduction

# Voice Activity Detection backends
webrtcvad>=2.0.10   # Google WebRTC VAD (fast, lightweight)
onnxruntime>=1.17.0  # For Silero VAD on 64-bit systems (aarch64)
# torch>=2.0.0       # PyTorch for Silero VAD (alternative, heavier)

# Audio playback (using MPV for robust playback)
python-mpv>=1.0.0

# Wake word detection (ONNX models; the playbook installs it with --no-deps)
openwakeword>=0.6.0

# Hardware control (Raspberry Pi)
spidev>=3.6  # SPI for APA102 LEDs
lgpio>=0.2.2.0  # GPIO for the button
gpiozero>=2.0  # LED power pin, display backlight

# Camera snapshots (hardware/camera.py)
Pillow>=10.0

# Configuration
pyyaml>=6.0

# Service discovery (optional)
zeroconf>=0.100.0

# System metrics (for monitoring)
psutil>=5.9.0

# BLE scanning (for presence detection, optional)
bleak>=0.22.0

# IRK-based RPA resolution (ble/rpa.py) — REQUIRED for phone presence; rpa.py
# silently no-ops (resolve_rpa() always returns False) if this is missing.
cryptography>=42.0.0

# Binary Opus audio transport (C1, voice-identity design) — OPTIONAL: only
# used when audio.codec: opus is configured; needs the libopus0 system
# package. Missing → the satellite logs a warning and stays on PCM.
opuslib>=3.0.1
